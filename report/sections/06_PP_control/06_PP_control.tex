        },
        sum/.style={
            draw,
            circle,

            inner sep=0pt
        },
        line/.style={->, thick},
        signal/.style={thick},
        branch/.style={circle, fill=black, inner sep=1.15pt}
    ]

    % =========================================================
    % Main blocks
    % =========================================================

    %Reference input blocks
    \node (xref) at (-2.5,-0.58) {$x_{ref}$};
    \node[sum] (sumx) at (-1.15,-0.58) {};
    \node[smallblock] (int) at (0.15,-0.58) {$\dfrac{1}{s}$};

    \node[mux] (mux) at (3.0,0) {MUX};

    % Manual triangular gain, pointing right
    \coordinate (Kin)  at (4.15,0);

    \node at (4.48,0) {$K_{\mathrm{ppi}}$};

    \node[sum] (sumi) at (5.80,0) {};
    \node[smallblock] (ie) at (5.80,0.95) {$i_e$};
    \node[block] (sat) at (7.5,0) {Current\\sat.};
    \node[smallblock] (Hi) at (9.5,0) {$H_i(s)$};
    \node[smallblock] (Gx) at (11.5,0) {$G_x(s)$};

    % Observer block
    \node[obsblock] (obs) at (6.30,+3.0) {State\\Observer};

    % =========================================================
    % Sum signs
    % =========================================================

    \node at ($(sumx.west)+(-0.1,0.2)$) {$+$};
    \node at ($(sumx.south)+(-0.2,-0.1)$) {$-$};

    \node at ($(sumi.north)+(0.15,+0.10)$) {$+$};
    \node at ($(sumi.west)+(-0.10,-0.15)$) {$+$};
    %\node at ($(sumi)$) {$+$};

    % =========================================================
    % Forward path
    % =========================================================

    \draw[line] (xref) -- (sumx);
    \draw[line] (sumx) -- node[above] {$e_x$} (int);

    \draw[line] (mux.east) -- node[above] {$x_{\mathrm{aug}}$} (Kin);
    \draw[line] (Kout) -- node[above] {$i_{PP}$} (sumi);
    \draw[line] (ie) -- (sumi);

    \draw[line] (sumi) -- node[above] {$i_{ref}$} (sat);
    \draw[line] (sat) -- (Hi);
    \draw[line] (Hi) -- node[above] {$i$} (Gx);
    \draw[line] (Gx.east) -- ++(1.1,0) coordinate (xout) node[right] {$x$};

    % =========================================================
    % Signal taps
    % =========================================================

    \coordinate (itap) at ($(Hi.east)!0.50!(Gx.west)$);
    \node[branch] at (itap) {};

    \coordinate (xtap) at ($(Gx.east)+(0.45,0)$);
    \node[branch] at (xtap) {};

    % =========================================================
    % Position feedback to error
    % routed above everything
    % =========================================================

    \draw[signal] (xtap) |- (-1.15,-2.35);
    \draw[line] (-1.15,-2.35) -- (sumx.south);

    % =========================================================
    % MUX input coordinates
    % order: x, xdot_hat, i, xi
    % =========================================================

    \coordinate (muxX)   at ($(mux.west)+(0,0.58)$);
    \coordinate (muxXd)  at ($(mux.west)+(0,0.20)$);
    \coordinate (muxI)   at ($(mux.west)+(0,-0.20)$);
    \coordinate (muxXi)  at ($(mux.west)+(0,-0.58)$);

    \node[left, xshift=-0.1cm,yshift=0.2cm] at (muxX)  {$x$};
    \node[left, xshift=-0.1cm,yshift=0.2cm] at (muxXd) {$\hat{\dot{x}}$};
    \node[left, xshift=-0.1cm,yshift=0.2cm] at (muxI)  {$i$};
    \node[left, xshift=-0.1cm,yshift=0.2cm] at (muxXi) {$x_i$};

    % =========================================================
    % xi from integrator to MUX
    % separate left vertical lane
    % =========================================================

    \draw[signal] (int.east) -- ++(0.35,0) coordinate (xiLaneTop);
    \draw[signal] (xiLaneTop) |- ($(muxXi)+(-0.45,0)$);
    \draw[line] ($(muxXi)+(-0.45,0)$) -- (muxXi);

    % =========================================================
    % Current measured bus
    % horizontal lower line
    % =========================================================

    \draw[signal] (itap) |- (8.5,-1.5);
    \draw[signal] (8.5,-1.5) -- (1.45,-1.5);
    \draw[signal] (1.45,-1.5) |- ($(muxI)+(-0.28,0)$);
    \draw[line] ($(muxI)+(-0.28,0)$) -- (muxI);

    % =========================================================
    % Position state
    % horizontal upper line
    % =========================================================

    \draw[signal] (xtap) |- (10.55,+1.7);
    \draw[signal] (10.55,+1.7) -- (2.0,+1.7);
    \draw[signal] (2.0,+1.7) |- ($(muxX)+(-0.5,0)$);
    \draw[line] ($(muxX)+(-0.5,0)$) -- (muxX);

    % =========================================================
    % Observer inputs from measured current and position
    % =========================================================

    % Current
    \coordinate (obs_i_in)  at ($(obs.east)+(0,0.22)$);
    \coordinate (obs_i_out) at ($(obs_i_in)+(0.35,0)$);

    \draw[signal] (itap) |- (obs_i_out);
    \draw[line] (obs_i_out) -- (obs_i_in);
    \node[right, xshift=0.05cm, yshift=0.2cm] at (obs_i_out) {$i$};

    % Position
    \coordinate (obs_x_in)  at ($(obs.east)+(0,-0.22)$);
    \coordinate (obs_x_out) at ($(obs_x_in)+(0.35,0)$);

    \draw[signal] (xtap) |- (obs_x_out);
    \draw[line] (obs_x_out) -- (obs_x_in);
    \node[right, xshift=0.05cm, yshift=0.2cm] at (obs_x_out) {$x$};

    % =========================================================
    % Observer outputs
    % x_hat shown but not used by PP
    % xdot_hat sent to the MUX
    % =========================================================

    % estimated postion
    \draw[line] ($(obs.west)+(0,0.22)$) -- ++(-0.65,0)
        node[left] {$\hat{x}$};

    % estimated velocity
    \draw[signal] ($(obs.west)+(0,-0.22)$) -- ++(-4.0,0)
        coordinate (xdLaneLow)
        node[midway, below] {$\hat{\dot{x}}$};

    \draw[signal] (xdLaneLow) |- ($(muxXd)+(-0.55,0)$);
    \draw[line] ($(muxXd)+(-0.55,0)$) -- (muxXd);

    % =========================================================
    % State definition
    % =========================================================

    \node[align=center] at (3.30,-3.35)
    {$x_{\mathrm{aug}} =
    \begin{bmatrix}
    x & \hat{\dot{x}} & i & x_i
    \end{bmatrix}^{T}$,
    \qquad
    $x_i = \displaystyle \int (x_{ref}-x)\,dt$};

    \end{tikzpicture}
    }
